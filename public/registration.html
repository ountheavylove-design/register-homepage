const API_URL = window.REGISTER_API_URL || "/api/register"; const form = document.getElementById('registerForm'); const msg = document.getElementById('message'); const btn = document.getElementById('submitBtn'); const homeBtn = document.getElementById('homeBtn'); const HOME_URL = window.REGISTER_HOME_URL || '/t24h/'; function setMessage(text, type) { msg.textContent = text; msg.className = 'msg ' + type; } homeBtn.addEventListener('click', () => { window.location.href = HOME_URL; }); form.addEventListener('submit', async (e) => { e.preventDefault(); const data = new FormData(form); const payload = { name: (data.get('name') || '').trim(), phone: (data.get('phone') || '').trim(), }; if (!payload.name || !payload.phone) { setMessage('Please fill in all fields.', 'error'); return; } btn.disabled = true; setMessage('Submitting...', ''); try { const res = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload), }); const result = await res.json(); if (!res.ok) { setMessage(result.error || 'Submission failed.', 'error'); } else { setMessage('Registered successfully!', 'success'); form.reset(); homeBtn.style.display = 'block'; } } catch (err) { setMessage('Network error. Please try again.', 'error'); } finally { btn.disabled = false; } });
